function(nl_get out)
  string(JSON value ERROR_VARIABLE error GET "${NL_JSON}" ${ARGN})
  if(error)
    message(FATAL_ERROR "Missing lock field: ${ARGN}: ${error}")
  endif()
  set(${out} "${value}" PARENT_SCOPE)
endfunction()
function(nl_approved)
  nl_get(ok ${ARGN} approved)
  nl_get(reference ${ARGN} reference)
  string(LENGTH "${reference}" reference_length)
  if(NOT ok OR NOT reference MATCHES "^[a-f0-9]+$" OR NOT reference_length EQUAL 40)
    message(FATAL_ERROR "Pending fixed approval: ${ARGN}")
  endif()
endfunction()
function(nl_pin out)
  nl_get(p ${ARGN} path)
  nl_get(real ${ARGN} realpath)
  nl_get(bytes ${ARGN} bytes)
  nl_get(expected ${ARGN} sha256)
  if(bytes LESS 0 OR (bytes EQUAL 0 AND NOT NL_ALLOW_EMPTY_PIN))
    message(FATAL_ERROR "Invalid empty executable/license/manifest input: ${ARGN}")
  endif()
  if(NOT IS_ABSOLUTE "${p}" OR NOT EXISTS "${p}" OR IS_DIRECTORY "${p}")
    message(FATAL_ERROR "Missing absolute input: ${ARGN}")
  endif()
  file(REAL_PATH "${p}" actual_real)
  file(SIZE "${p}" actual_bytes)
  file(SHA256 "${p}" actual_sha)
  if(NOT actual_real STREQUAL real OR NOT actual_bytes STREQUAL bytes OR NOT actual_sha STREQUAL expected)
    message(FATAL_ERROR "Pinned input drift: ${ARGN}")
  endif()
  set(${out} "${p}" PARENT_SCOPE)
endfunction()
# Supply inventories are fixed metadata inputs, not fabricated post-build deps.
function(nl_inventory required_root)
  nl_pin(manifest ${ARGN})
  file(READ "${manifest}" NL_JSON)
  string(JSON roots LENGTH "${NL_JSON}" roots)
  string(JSON count LENGTH "${NL_JSON}" files)
  if(roots LESS 1 OR count LESS 1 OR count GREATER 50000)
    message(FATAL_ERROR "Invalid supply inventory size")
  endif()
  math(EXPR root_last "${roots}-1")
  math(EXPR file_last "${count}-1")
  set(required_covered FALSE)
  foreach(j RANGE 0 ${root_last})
    nl_get(r roots ${j})
    if(NOT IS_ABSOLUTE "${r}")
      message(FATAL_ERROR "Supply inventory root must be absolute")
    endif()
    if(r STREQUAL required_root)
      set(required_covered TRUE)
    endif()
  endforeach()
  if(NOT required_root STREQUAL "" AND NOT required_covered)
    message(FATAL_ERROR "Inventory does not cover selected SDK root")
  endif()
  set(seen "")
  set(NL_ALLOW_EMPTY_PIN TRUE)
  foreach(i RANGE 0 ${file_last})
    nl_pin(p files ${i})
    nl_get(real files ${i} realpath)
    if(p IN_LIST seen)
      message(FATAL_ERROR "Duplicate inventory path")
    endif()
    list(APPEND seen "${p}")
    set(covered FALSE)
    foreach(j RANGE 0 ${root_last})
      nl_get(r roots ${j})
      cmake_path(IS_PREFIX r "${p}" NORMALIZE in_root)
      cmake_path(IS_PREFIX r "${real}" NORMALIZE real_in_root)
      if(in_root AND real_in_root AND NOT p STREQUAL r AND NOT real STREQUAL r)
        set(covered TRUE)
      endif()
    endforeach()
    if(NOT covered)
      message(FATAL_ERROR "Inventory input escapes its approved roots")
    endif()
  endforeach()
endfunction()

    endif()

  file(SHA256 "${p}" actual_sha)
  if(NOT bytes STREQUAL actual_bytes OR NOT expected_sha STREQUAL actual_sha)
    message(FATAL_ERROR "Fixed source drift: ${relative}")
  endif()

function(nl_target target)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
  set_target_properties(${target} PROPERTIES
    CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF
    POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON
    RUNTIME_OUTPUT_DIRECTORY "${NL_ARTIFACTS}"
    LIBRARY_OUTPUT_DIRECTORY "${NL_ARTIFACTS}"
    ARCHIVE_OUTPUT_DIRECTORY "${NL_ARTIFACTS}"
    PDB_OUTPUT_DIRECTORY "${NL_ARTIFACTS}"
    RUNTIME_OUTPUT_DIRECTORY_RELWITHDEBINFO "${NL_ARTIFACTS}"
    LIBRARY_OUTPUT_DIRECTORY_RELWITHDEBINFO "${NL_ARTIFACTS}"
    ARCHIVE_OUTPUT_DIRECTORY_RELWITHDEBINFO "${NL_ARTIFACTS}")
  target_link_libraries(${target} PRIVATE ${NL_SYSTEM_LIBRARIES})
endfunction()
