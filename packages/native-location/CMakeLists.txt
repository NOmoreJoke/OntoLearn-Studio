function(nl_get out)
  string(JSON value ERROR_VARIABLE error GET "${NL_JSON}" ${ARGN})
  if(error)
    message(FATAL_ERROR "Missing lock field: ${ARGN}: ${error}")
  endif()
  set(${out} "${value}" PARENT_SCOPE)
endfunction()
function(nl_approved)
  nl_get(ok ${ARGN} approved)
  nl_get(reference ${ARGN} reference)
  string(LENGTH "${reference}" reference_length)
  if(NOT ok OR NOT reference MATCHES "^[a-f0-9]+$" OR NOT reference_length EQUAL 40)
    message(FATAL_ERROR "Pending fixed approval: ${ARGN}")
  endif()
endfunction()
function(nl_pin out)
  nl_get(p ${ARGN} path)
  nl_get(real ${ARGN} realpath)
  nl_get(bytes ${ARGN} bytes)
  nl_get(expected ${ARGN} sha256)
  if(bytes LESS 0 OR (bytes EQUAL 0 AND NOT NL_ALLOW_EMPTY_PIN))
    message(FATAL_ERROR "Invalid empty executable/license/manifest input: ${ARGN}")
  endif()
  if(NOT IS_ABSOLUTE "${p}" OR NOT EXISTS "${p}" OR IS_DIRECTORY "${p}")
    message(FATAL_ERROR "Missing absolute input: ${ARGN}")
  endif()
  file(REAL_PATH "${p}" actual_real)
  file(SIZE "${p}" actual_bytes)
  file(SHA256 "${p}" actual_sha)
  if(NOT actual_real STREQUAL real OR NOT actual_bytes STREQUAL bytes OR NOT actual_sha STREQUAL expected)
    message(FATAL_ERROR "Pinned input drift: ${ARGN}")
  endif()
  set(${out} "${p}" PARENT_SCOPE)
endfunction()
# Non-executable small records reconstruct the original pinned JSON byte for byte.
# Only these flat canonical key layouts are accepted; JSON handles string decoding.
# Lossless CMake-list transport only; this does not decode JSON strings.
function(nl_record_decode out encoded)
  string(REPLACE "@S" ";" decoded "${encoded}")
  string(REPLACE "@R" "]" decoded "${decoded}")
  string(REPLACE "@L" "[" decoded "${decoded}")
  string(REPLACE "@B" "\\" decoded "${decoded}")
  string(REPLACE "@A" "@" decoded "${decoded}")
  set(${out} "${decoded}" PARENT_SCOPE)
endfunction()
function(nl_inventory_metadata out_records required_root)
  foreach(kind source records)
    nl_get(metadata_bytes ${ARGN} ${kind} bytes)
    nl_get(metadata_path ${ARGN} ${kind} path)
    if(metadata_bytes LESS 1 OR metadata_bytes GREATER 16777216 OR
       NOT EXISTS "${metadata_path}" OR IS_DIRECTORY "${metadata_path}")
      message(FATAL_ERROR "Supply metadata exceeds 16MiB")
    endif()
    file(SIZE "${metadata_path}" actual_metadata_bytes)
    if(NOT actual_metadata_bytes EQUAL metadata_bytes OR actual_metadata_bytes GREATER 16777216)
      message(FATAL_ERROR "Supply metadata size drift/cap")
    endif()
  endforeach()
  nl_get(source_sha ${ARGN} source sha256)
  nl_get(records_sha ${ARGN} records sha256)
  nl_pin(source ${ARGN} source)
  nl_pin(records ${ARGN} records)
  file(READ "${source}" NL_JSON)
  string(JSON source_type TYPE "${NL_JSON}")
  string(JSON roots_type TYPE "${NL_JSON}" roots)
  string(JSON files_type TYPE "${NL_JSON}" files)
  string(JSON source_fields LENGTH "${NL_JSON}")
  string(JSON roots LENGTH "${NL_JSON}" roots)
  string(JSON count LENGTH "${NL_JSON}" files)
  if(NOT source_type STREQUAL "OBJECT" OR NOT roots_type STREQUAL "ARRAY" OR
     NOT files_type STREQUAL "ARRAY" OR NOT source_fields EQUAL 2 OR
     roots LESS 1 OR roots GREATER 64 OR count LESS 1 OR count GREATER 50000)
    message(FATAL_ERROR "Invalid supply inventory size/fields")
  endif()
  math(EXPR root_last "${roots}-1")
  set(required_covered FALSE)
  foreach(j RANGE 0 ${root_last})
    nl_get(r roots ${j})
    string(JSON root_type TYPE "${NL_JSON}" roots ${j})
    if(NOT root_type STREQUAL "STRING" OR NOT IS_ABSOLUTE "${r}" OR r MATCHES "[;\r\n]")
      message(FATAL_ERROR "Supply inventory root must be absolute text")
    endif()
    set(approved_root_${j} "${r}")
    if(r STREQUAL required_root)
      set(required_covered TRUE)
    endif()
  endforeach()
  if(NOT required_root STREQUAL "" AND NOT required_covered)
    message(FATAL_ERROR "Inventory does not cover selected SDK root")
  endif()
  file(READ "${records}" raw_records)
  string(REPLACE "@" "@A" encoded "${raw_records}")
  string(REPLACE "\\" "@B" encoded "${encoded}")
  string(REPLACE "[" "@L" encoded "${encoded}")
  string(REPLACE "]" "@R" encoded "${encoded}")
  string(REPLACE ";" "@S" encoded "${encoded}")
  string(REPLACE "\n" ";" lines "${encoded}")
  unset(raw_records)
  unset(encoded)
  list(POP_BACK lines trailing)
  if(NOT trailing STREQUAL "")
    message(FATAL_ERROR "Inventory records require a final LF")
  endif()
  list(LENGTH lines line_count)
  math(EXPR expected_lines "${count}+1")
  if(NOT line_count EQUAL expected_lines)
    message(FATAL_ERROR "Inventory record count differs")
  endif()
  list(POP_FRONT lines encoded_header)
  nl_record_decode(header "${encoded_header}")
  set(json_string [=["([^"\\]|\\.)*"]=])
  set(header_pattern "^\\{\"format\":\"n0-inventory-records-v1\",\"count\":[1-9][0-9]*,\"roots\":\\[${json_string}(,${json_string})*\\],\"prefix\":${json_string},\"suffix\":${json_string}\\}$")
  string(LENGTH "${header}" header_bytes)
  if(header_bytes GREATER 8192 OR NOT header MATCHES "${header_pattern}")
    message(FATAL_ERROR "Invalid inventory header grammar")
  endif()
  set(NL_JSON "${header}")
  nl_get(header_count count)
  string(JSON header_roots LENGTH "${NL_JSON}" roots)
  if(NOT header_count EQUAL count OR NOT header_roots EQUAL roots)
    message(FATAL_ERROR "Inventory header count/roots differ")
  endif()
  foreach(j RANGE 0 ${root_last})
    nl_get(r roots ${j})
    set(expected_root "${approved_root_${j}}")
    if(NOT r STREQUAL expected_root)
      message(FATAL_ERROR "Inventory header roots differ")
    endif()
  endforeach()
  nl_get(prefix prefix)
  nl_get(suffix suffix)
  # Prove the record slot is precisely the files array, using the JSON parser.
  set(stub "${prefix}\"N0-record-slot\"${suffix}")
  string(JSON stub_fields LENGTH "${stub}")
  string(JSON stub_files TYPE "${stub}" files)
  string(JSON stub_count LENGTH "${stub}" files)
  string(JSON stub_slot GET "${stub}" files 0)
  string(JSON stub_roots LENGTH "${stub}" roots)
  if(NOT stub_fields EQUAL 2 OR NOT stub_files STREQUAL "ARRAY" OR
     NOT stub_count EQUAL 1 OR NOT stub_slot STREQUAL "N0-record-slot" OR NOT stub_roots EQUAL roots)
    message(FATAL_ERROR "Inventory record slot is not the files array")
  endif()
  foreach(j RANGE 0 ${root_last})
    string(JSON r GET "${stub}" roots ${j})
    set(expected_root "${approved_root_${j}}")
    if(NOT r STREQUAL expected_root)
      message(FATAL_ERROR "Inventory record slot roots differ")
    endif()
  endforeach()
  list(JOIN lines "," encoded_joined)
  nl_record_decode(joined "${encoded_joined}")
  string(SHA256 reconstructed_sha "${prefix}${joined}${suffix}")
  if(NOT reconstructed_sha STREQUAL source_sha)
    message(FATAL_ERROR "Inventory source reconstruction differs")
  endif()
  # Canonical reconstruction rejects BOM, CR, missing/extra LF and transport drift.
  list(JOIN lines "\n" encoded_joined)
  nl_record_decode(joined "${encoded_joined}")
  string(SHA256 canonical_records_sha "${header}\n${joined}\n")
  if(NOT canonical_records_sha STREQUAL records_sha)
    message(FATAL_ERROR "Inventory record encoding/line endings differ")
  endif()
  unset(joined)
  unset(encoded_joined)
  set(record_pattern "^\\{\"path\":${json_string},\"realpath\":${json_string},\"bytes\":(0|[1-9][0-9]*),\"sha256\":\"[a-f0-9]+\"\\}$")
  foreach(encoded_line IN LISTS lines)
    nl_record_decode(line "${encoded_line}")
    string(LENGTH "${line}" line_bytes)
    if(line_bytes GREATER 4096 OR NOT line MATCHES "${record_pattern}")
      message(FATAL_ERROR "Invalid inventory record grammar")
    endif()
    set(NL_JSON "${line}")
    nl_get(p path)
    nl_get(real realpath)
    nl_get(expected sha256)
    string(LENGTH "${expected}" sha_length)
    if(NOT sha_length EQUAL 64 OR NOT IS_ABSOLUTE "${p}" OR NOT IS_ABSOLUTE "${real}" OR
       p MATCHES "[;\r\n]" OR real MATCHES "[;\r\n]")
      message(FATAL_ERROR "Invalid inventory pin text/hash")
    endif()
    string(SHA256 key "${p}")
    if(DEFINED NL_SEEN_${key})
      message(FATAL_ERROR "Duplicate inventory path or digest-key collision")
    endif()
    set(NL_SEEN_${key} "${p}")
    set(covered FALSE)
    foreach(j RANGE 0 ${root_last})
      set(r "${approved_root_${j}}")
      cmake_path(IS_PREFIX r "${p}" NORMALIZE in_root)
      cmake_path(IS_PREFIX r "${real}" NORMALIZE real_in_root)
      if(in_root AND real_in_root AND NOT p STREQUAL r AND NOT real STREQUAL r)
        set(covered TRUE)
      endif()
    endforeach()
    if(NOT covered)
      message(FATAL_ERROR "Inventory input escapes its approved roots")
    endif()
  endforeach()
  set(${out_records} "${lines}" PARENT_SCOPE)
endfunction()
function(nl_inventory required_root)
  nl_inventory_metadata(records "${required_root}" ${ARGN})
  list(LENGTH records count)
  message(STATUS "N0 inventory metadata mapped: ${count}; validating every physical pin")
  set(NL_ALLOW_EMPTY_PIN TRUE)
  foreach(encoded_line IN LISTS records)
    nl_record_decode(NL_JSON "${encoded_line}")
    nl_pin(p)
  endforeach()
  message(STATUS "N0 inventory physical pins complete: ${count}")
endfunction()

    CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF

    LIBRARY_OUTPUT_DIRECTORY "${NL_ARTIFACTS}"
    ARCHIVE_OUTPUT_DIRECTORY "${NL_ARTIFACTS}"
    PDB_OUTPUT_DIRECTORY "${NL_ARTIFACTS}"

    LIBRARY_OUTPUT_DIRECTORY_RELWITHDEBINFO "${NL_ARTIFACTS}"
