                 0o600, dir_fd=dirfd)

def pin_pass():
    rows = []
    used = 0
    for pin in (CORE_PIN, PROBE_PIN):
        if (set(pin) != {"path", "bytes", "sha256"} or type(pin["bytes"]) is not int
                or not 0 < pin["bytes"] <= ARTIFACT_LIMIT
                or type(pin["path"]) is not str or not pin["path"].startswith("/")
                or len(pin["path"]) > 4096 or "\x00" in pin["path"]
                or type(pin["sha256"]) is not str or len(pin["sha256"]) != 64
                or any(c not in "0123456789abcdef" for c in pin["sha256"])):
            raise ValueError("unresolved-pin")
        used += pin["bytes"]
        if used > ARTIFACT_LIMIT:
            raise ValueError("artifact-budget")
        fd = os.open(pin["path"], os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC)
        try:
            before = os.fstat(fd)
            if (not stat.S_ISREG(before.st_mode) or before.st_size != pin["bytes"]
                    or not before.st_mode & 0o111):
                raise ValueError("artifact-kind-size-mode")
            digest = hashlib.sha256()
            remaining = pin["bytes"]
            while remaining:
                block = os.read(fd, min(65536, remaining))
                if not block:
                    raise ValueError("artifact-short-read")
                digest.update(block)
                remaining -= len(block)
            after = os.fstat(fd)
            named = os.stat(pin["path"], follow_symlinks=False)
            identity = lambda s: (s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns, s.st_ctime_ns)
            if identity(before) != identity(after) or identity(after) != identity(named):
                raise ValueError("artifact-drift")
            if digest.hexdigest() != pin["sha256"]:
                raise ValueError("artifact-hash")
            rows.append({**pin, "dev": after.st_dev, "ino": after.st_ino})
        finally:
            os.close(fd)
    return rows

def wait_until(deadline, allow_cancel=True):
    global owned_unreaped, process_api_refused
    while time.monotonic() < deadline:
        if allow_cancel and cancel_signal:
            return "cancelled"
        try:
            pid, status = os.waitpid(proc.pid, os.WNOHANG)
        except BaseException as exc:
            # Python层收到ECHILD/EINTR/权限等异常即停；CPython可能已按PEP475内部重试EINTR。
            # 不伪造exit0，不再依号码发信号，不在本脚本重试该异常API。
            owned_unreaped = False
            process_api_refused = True
            event("waitpid-stopped-unknown", exc)
            return "wait-unknown"
        if pid == proc.pid:
            if os.WIFEXITED(status):
                code = os.WEXITSTATUS(status)
            elif os.WIFSIGNALED(status):
                code = -os.WTERMSIG(status)
            else:
                owned_unreaped = False
                event("unexpected-nonterminal-wait-status")
                return "wait-unknown"
            owned_unreaped = False
            proc.returncode = code  # 本控制器唯一exact wait；避免Popen内部另一轮wait。
            result.update({"coreReturncode": code, "coreWaitStatus": status})
            return "terminal"
        if pid != 0:
            owned_unreaped = False
            event("unexpected-wait-pid")
            return "wait-unknown"
        time.sleep(min(0.01, max(0.0, deadline - time.monotonic())))
    return "deadline"

def stream_pass(fd, name):
    before = os.fstat(fd)
    if not stat.S_ISREG(before.st_mode) or before.st_size > STREAM_LIMIT:
        return {"name": name, "bytes": before.st_size, "sha256": None, "stable": False}
    digest = hashlib.sha256()
    remaining = before.st_size
    offset = 0
    while remaining:
        block = os.pread(fd, min(65536, remaining), offset)  # 不移动与child共享的open-file offset。
        if not block:
            raise ValueError("raw-stream-short-read")
        digest.update(block)
        remaining -= len(block)
        offset += len(block)
    after = os.fstat(fd)
    stable = (before.st_size, before.st_mtime_ns, before.st_ctime_ns) == (
        after.st_size, after.st_mtime_ns, after.st_ctime_ns)
    return {"name": name, "bytes": before.st_size, "sha256": digest.hexdigest(), "stable": stable}
