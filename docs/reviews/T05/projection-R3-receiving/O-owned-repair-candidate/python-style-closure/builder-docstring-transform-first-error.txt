def canonical(value):
    # ASCII preserves intentional isolated-surrogate rejection inputs losslessly.
    return json.dumps(
        value, ensure_ascii=True, sort_keys=True, separators=(",", ":"), allow_nan=False
    ).encode()


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def strict_json(raw):
    def pairs(items):
        result = {}
        for key, value in items:
            if key in result:
                raise ValueError("duplicate JSON key: " + key)
            result[key] = value
        return result

    return json.loads(
        raw, object_pairs_hook=pairs, parse_constant=lambda x: (_ for _ in ()).throw(ValueError(x))
    )


def read_pinned(name):
    raw = (INPUTS / name).read_bytes()
    size, sha = PINS[name]
    if len(raw) != size or digest(raw) != sha:
        raise ValueError("fixed input identity mismatch: " + name)
    return strict_json(raw)

def checked_rule_ids(values):
    """Validate finite declaration metadata; this does not construct RuleSpec."""
    if type(values) is not list or len(values) > 100:
        raise ValueError("complete rule ID list required; at most100 definitions")
    for value in values:
        if type(value) is not str or not value or len(value) > 16384:
            raise ValueError("rule ID must be a nonempty exact string within character cap")
        try:
            size = len(value.encode("utf-8"))
        except UnicodeEncodeError as error:
            raise ValueError("rule ID must be valid Unicode") from error
        if size > 16384:
            raise ValueError("rule ID UTF8 cap")
    if len(set(values)) != len(values):
        raise ValueError("duplicate rule ID")
    return values


def declared_allocation_expectation(rule_ids, disabled_rule_ids):
    """Pure seven-pool relation on a complete explicitly supplied ID universe.

    Empty input has the approved anonymous dummy slot. Its original empty batch
    must still be captured; no actual object, remaining ten limits, or usage is
    constructed here. Missing metadata is not equivalent to an empty list.
    """
    checked_rule_ids(rule_ids)
    checked_rule_ids(disabled_rule_ids)
    if not set(disabled_rule_ids).issubset(rule_ids):
        raise ValueError("disabled ID outside complete universe")
    n = max(1, len(rule_ids))
    children = []
    for j, rule_id in enumerate(sorted(rule_ids) if rule_ids else [None]):
        child = {
            "ruleId": rule_id,
            "definitionOrdinal": rule_ids.index(rule_id) if rule_ids else None,
            "sortedPosition": j,
            "declaredLimits": {
                key: total // n + int(j < total % n) for key, total in DECLARED_POOLS.items()
            },
        }
        if rule_ids:
            child["disabled"] = rule_id in disabled_rule_ids
        else:
            child["dummyContract"] = {
                "definitionCount": 0,
                "originalMatchCount": 0,
                "requiresRealEmptyBatch": True,
                "dummyDisabledFlag": "not specified by cited paragraphs; not invented",
            }
        children.append(child)
    return {
        "format": "T05-declared-allocation-expectation/1",
        "namespace": "declared-only",
        "policyVersion": "i05-profile-graph-budget/1",
        "originalRuleIds": list(rule_ids),
        "disabledRuleIds": sorted(disabled_rule_ids),
        "definitionCount": len(rule_ids),

        "compileEvaluateSameBudgetRequired": True,
        "resetOrReallocationPermitted": False,
        "actualAllocationCaptureProvided": False,

        return {"status": "not-invoked", "reason": "structured-material-required before allocation"}
    raw = dataset.get("input", dataset)
    key = "definitions" if dataset["kind"] == "profile" else "rules"
    if key not in raw or type(raw[key]) is not list:
        return {
            "status": "pending",

            "reason": "Turtle graph is not an explicit complete RuleSpec universe",
            "expectation": None,
        }
    definitions = raw[key]
    if not all(type(item) is dict and "rule_id" in item for item in definitions):
        return {
            "status": "pending",
            "missingFields": ["valid complete definition ID metadata"],
            "expectation": None,
        }
    rule_ids = [item["rule_id"] for item in definitions]
    try:
        checked_rule_ids(rule_ids)
    except ValueError as error:
        return {
            "status": "pending",
            "missingFields": ["valid unique complete rule IDs"],
            "reason": str(error),
            "expectation": None,
        }
    if "disabled_rule_ids" not in raw:
        return {
            "status": "pending",
            "knownRuleUniverse": [
                {"ruleId": name, "definitionOrdinal": i} for i, name in enumerate(rule_ids)

            "missingFields": ["explicit complete disabled_rule_ids; omitted is not declared empty"],

def iri(value):
    return {"kind": "iri", "value": value}


def literal(value, datatype=None, language=None):
    return {
        "kind": "literal",
        "lexical": value,
        "datatype": RDF + "langString" if language else datatype or XSD + "string",
        "language": language.lower() if language else None,
    }


def term_key(term):
    return canonical(term)


def ordered_set(values):
    return [json.loads(k) for k in sorted({canonical(v) for v in values})]


def n3_term(token):
    if token.startswith("<") and token.endswith(">"):
        return iri(token[1:-1])
    match = re.fullmatch(r'("(?:\\.|[^"\\])*")(?:(@[^ ]+)|\^\^<([^>]+)>)?', token, re.S)
    if match:
        return literal(json.loads(match[1]), match[3], match[2][1:] if match[2] else None)
    raise ValueError("unsupported independent N3 token: " + token)


class FixtureTurtle:
    """Decode only the explicit finite Turtle grammar used by valid sealed fixtures."""

    token = re.compile(
        (
            "\\s+|\\#[^\\n]*|<[^>]*>|[+-]?(?:\\d+\\.\\d+|\\.\\d+|\\d+)[eE][+-]?\\d+|[+-]?(?:\\d"
            '+\\.\\d+|\\.\\d+)|"(?:\\\\.|[^"\\\\])*"|\\^\\^|@[\\w-]+|[.;,()\\[\\]]|[^\\s.;,()\\'
            '[\\]<>"^]+'
        ),
        re.S,
    )

    def __init__(self, text, *, allow_shape_lists=False):
        self.allow_shape_lists = allow_shape_lists
        self.tokens = []
        position = 0
        for match in self.token.finditer(text):
            if match.start() != position:
                raise ValueError("fixture token gap")
            position = match.end()
            token = match[0]
            if not token.isspace() and not token.startswith("#"):
                self.tokens.append(token)
        if position != len(text):
            raise ValueError("fixture trailing token gap")
        self.i = 0
        self.prefix = {"rdf": RDF, "rdfs": RDFS, "xsd": XSD, "sh": SH, "owl": OWL}
        self.graph = []
        self.blank = 0

    def pop(self, expected=None):
        if self.i == len(self.tokens):
            raise ValueError("fixture unexpectedly ended")
        token = self.tokens[self.i]
        self.i += 1
        if expected is not None and token != expected:
            raise ValueError("fixture expected " + expected + " got " + token)
        return token

    def peek(self):
        return self.tokens[self.i] if self.i < len(self.tokens) else None

    def term(self):
        token = self.pop()
        if token == "(":
            if not self.allow_shape_lists:
                raise ValueError("fixture list blank nodes are admitted only in the shapes role")
            items = []
            while self.peek() != ")":
                items.append(self.term())
            self.pop(")")
            tail = iri(RDF + "nil")
            for item in reversed(items):
                self.blank += 1
                node = {"kind": "blank", "label": "list-" + str(self.blank)}
                self.graph += [[node, iri(RDF + "first"), item], [node, iri(RDF + "rest"), tail]]
                tail = node
            return tail
        if token.startswith("<"):
            return iri(token[1:-1])
        if token.startswith('"'):
            text = json.loads(token)
            datatype, language = None, None
            if self.peek() == "^^":
                self.pop()
                datatype = self.term()["value"]
            elif self.peek() and self.peek().startswith("@"):
                language = self.pop()[1:]
            return literal(text, datatype, language)
        if token == "a":
            return iri(RDF + "type")
        if token in ("true", "false"):
            return literal(token, XSD + "boolean")
        if re.fullmatch(r"[+-]?\d+", token):
            return literal(token, XSD + "integer")
        if re.fullmatch(r"[+-]?(?:\d+\.\d+|\.\d+|\d+)[eE][+-]?\d+", token):
            return literal(token, XSD + "double")
        if re.fullmatch(r"[+-]?(?:\d+\.\d+|\.\d+)", token):
            return literal(token, XSD + "decimal")
        if ":" in token:
            prefix, value = token.split(":", 1)
            if prefix in self.prefix:
                return iri(self.prefix[prefix] + value)
        raise ValueError("unsupported finite Turtle term: " + token)

    def parse(self):
        while self.peek() is not None:
            if self.peek().lower() in ("@prefix", "prefix"):
                directive = self.pop().lower()
                prefix = self.pop()
                self.prefix[prefix[:-1]] = self.pop()[1:-1]
                if directive == "@prefix":
                    self.pop(".")
                continue
            subject = self.term()
            while True:
                predicate = self.term()
                while True:
                    self.graph.append([subject, predicate, self.term()])
                    if self.peek() != ",":
                        break
                    self.pop(",")
                if self.peek() != ";":
                    break
                self.pop(";")
                if self.peek() == ".":
                    break
            self.pop(".")
        return ordered_set(self.graph)


def turtle(text, *, allow_shape_lists=False):
    return FixtureTurtle(text, allow_shape_lists=allow_shape_lists).parse()


def values(graph, subject, predicate):
    return [o for s, p, o in graph if s == subject and p == iri(predicate)]


def scalar(term):
    if term["kind"] != "literal":
        return None
    datatype, lexical = term["datatype"], term["lexical"]
    if datatype in [
        XSD + n for n in ("integer", "int", "nonNegativeInteger", "decimal", "float", "double")
    ]:
        return Decimal(lexical)
    if datatype == XSD + "date":
        return date.fromisoformat(lexical[:-1])
    if datatype == XSD + "dateTime":
        return datetime.fromisoformat(lexical.replace("Z", "+00:00"))
    return None


def shacl_report(data, shapes):
    """Independent declarative evaluation of the six explicitly admitted templates."""
    rows = []

    def violation(focus, path, shape, component, value=None):
        rows.append(
            {
                "focus": focus,
                "path": path,
                "value": value,
                "sourceShape": shape,
                "constraint": iri(SH + component + "ConstraintComponent"),
                "severity": iri(SH + "Violation"),
            }
        )

    nodes = [s for s, p, o in shapes if p == iri(RDF + "type") and o == iri(SH + "NodeShape")]
    for node in nodes:
        (target,) = values(shapes, node, SH + "targetClass")
        focuses = [s for s, p, o in data if p == iri(RDF + "type") and o == target]
        for prop in values(shapes, node, SH + "property"):
            (path,) = values(shapes, prop, SH + "path")
            for focus in focuses:
                objects = values(data, focus, path["value"])
                for count_name, valid in [
                    ("minCount", lambda n, b: n >= b),
                    ("maxCount", lambda n, b: n <= b),
                ]:
                    for bound in values(shapes, prop, SH + count_name):
                        if not valid(len(objects), int(bound["lexical"])):
                            violation(focus, path, prop, count_name[0].upper() + count_name[1:])
                for value in objects:
                    for bound in values(shapes, prop, SH + "datatype"):
                        good = value["kind"] == "literal" and value["datatype"] == bound["value"]
                        if good and value["datatype"] == XSD + "integer":
                            good = re.fullmatch(r"[+-]?\d+", value["lexical"]) is not None
                        if not good:
                            violation(focus, path, prop, "Datatype", value)
                    for first in values(shapes, prop, SH + "in"):
                        allowed, current = [], first
                        while current != iri(RDF + "nil"):
                            (item,) = values(shapes, current, RDF + "first")
                            allowed.append(item)
                            (current,) = values(shapes, current, RDF + "rest")
                        if value not in allowed:
                            violation(focus, path, prop, "In", value)
                    for name, valid in [
                        ("minInclusive", lambda a, b: a >= b),
                        ("maxInclusive", lambda a, b: a <= b),
                    ]:
                        for bound in values(shapes, prop, SH + name):
                            try:
                                a, b = scalar(value), scalar(bound)
                                accepted = a is not None and b is not None and valid(a, b)
                            except (TypeError, ValueError, InvalidOperation):
                                accepted = False
                            if not accepted:
                                violation(focus, path, prop, name[0].upper() + name[1:], value)
    return {"conforms": not rows, "violations": sorted(rows, key=canonical)}


def graph_from_records(records):
    return ordered_set(
        [
            [
                iri(r["triple"]["subject"]),
                iri(r["triple"]["predicate"]),
                n3_term(r["triple"]["object"]),
            ]
            for r in records
        ]
    )


def profile_value(token):
    term = n3_term(token)
    if term["kind"] == "iri":
        return {"domain": "I", "value": term["value"], "rdfTerm": term}
    if term["datatype"] in (XSD + "integer", XSD + "int", XSD + "nonNegativeInteger"):
        return {
            "domain": "N",
            "value": {"numerator": int(term["lexical"]), "denominator": 1, "negativeZero": False},
            "rdfTerm": term,
        }
    if term["datatype"] == XSD + "boolean":
        return {"domain": "B", "value": term["lexical"] in ("true", "1"), "rdfTerm": term}
    if term["datatype"] in (XSD + "string", RDF + "langString"):
        return {
            "domain": "L" if term["language"] else "S",
            "value": term["lexical"],
            "rdfTerm": term,
        }
    raise ValueError("No independently declared profile scalar for this corpus term: " + token)

def finite_fixture_index(key, prefix, allowed):
    suffix = key.removeprefix(prefix)
    if not key.startswith(prefix) or not suffix.isascii() or not suffix.isdecimal():
        raise ValueError("unapproved finite fixture index: " + key)
    index = int(suffix)
    if str(index) != suffix or index not in allowed:
        raise ValueError("unapproved finite fixture index: " + key)
    return index

def finite_rdfs(graph):
    """Independent least fixed point of the four declared entailment relations."""
    initial = ordered_set(graph)
    closure = initial[:]
    while True:
        proposals = []
        for s, p, o in closure:
            if p == iri(RDFS + "subClassOf"):
                for x, q, y in closure:
                    if q == iri(RDFS + "subClassOf") and o == x:
                        proposals.append([s, p, y])
                    if q == iri(RDF + "type") and y == s:
                        proposals.append([x, q, o])
            if p in (iri(RDFS + "domain"), iri(RDFS + "range")):
                for x, q, y in closure:
                    if q == s:
                        subject = x if p == iri(RDFS + "domain") else y
                        if subject["kind"] == "iri":
                            proposals.append([subject, iri(RDF + "type"), o])
        revised = ordered_set(closure + proposals)
        if revised == closure:
            return [t for t in closure if t not in initial]
        closure = revised


def declared_rules(rules, graph):
    """Relational conjunctive-query interpretation; no product rule objects."""
    all_matches, heads = [], []
    for ordinal, rule in enumerate(rules):
        bindings = [({}, [])]
        for pattern in rule["when"]:
            joined = []
            for binding, when in bindings:
                for triple in graph:
                    trial = dict(binding)
                    for pattern_term, value in zip(pattern, triple):
                        if "variable" in pattern_term:
                            variable = pattern_term["variable"]
                            if variable in trial and trial[variable] != value:
                                break
                            trial[variable] = value
                        elif record_term(pattern_term) != value:
                            break
                    else:
                        joined.append((trial, when + [triple]))
            bindings = joined
        for binding, when in sorted(bindings, key=canonical):
            if rule.get("comparisons"):
                raise ValueError("legacy comparison needs separately authored finite expected")
            produced = []
            for index, pattern in enumerate(rule["heads"]):
                triple = [
                    binding[t["variable"]] if "variable" in t else record_term(t) for t in pattern
                ]
                heads.append(triple)
                produced.append({"headIndex": index, "triple": triple})
            all_matches.append(
                {
                    "ruleId": rule["rule_id"],
                    "definitionOrdinal": ordinal,
                    "bindings": dict(sorted(binding.items())),
                    "when": when,
                    "comparisons": [],
                    "accepted": True,
                    "heads": produced,
                }
            )
    return {"customOutput": ordered_set(heads), "matches": all_matches}


def record_term(term):
    if term["kind"] == "iri":
        return iri(term["lexical_form"])
    return literal(term["lexical_form"], term["datatype_iri"], term["language_raw"])


def key_n3(term):
    if term["kind"] == "iri":
        return "<" + term["value"] + ">"
    text = json.dumps(term["lexical"], ensure_ascii=False)
    if term["language"]:
        return text + "@" + term["language"]
    return text if term["datatype"] == XSD + "string" else text + "^^<" + term["datatype"] + ">"


def finite_witnesses(graph):
    """Four normative relation instances, sorted by the approved §2 pair key.

    This selects stable first insertion, never the shortest proof or all proofs.
    Logical leaf references deliberately do not invent producer leaf IDs.
    """    closure = ordered_set(graph)
    refs = {canonical(t): {"kind": "input-triple", "triple": t} for t in closure}
    depths = {canonical(t): 0 for t in closure}
    leaves = {canonical(t): [t] for t in closure}
    steps = []
    while True:
        candidates = []
        for a in closure:
            s, p, o = a
            for b in closure:
                x, q, y = b
                if p == iri(RDFS + "subClassOf"):
                    if q == iri(RDFS + "subClassOf") and o == x:
                        candidates.append(([s, p, y], [a, b], "rdfs11"))
                    if q == iri(RDF + "type") and y == s:
                        candidates.append(([x, q, o], [b, a], "rdfs9"))
                if p == iri(RDFS + "domain") and q == s:
                    candidates.append(([x, iri(RDF + "type"), o], [a, b], "rdfs2"))
                if p == iri(RDFS + "range") and q == s and y["kind"] == "iri":
                    candidates.append(([y, iri(RDF + "type"), o], [a, b], "rdfs3"))

        def sort_key(item):
            result, premises, _rule = item
            return tuple(map(key_n3, result)), tuple(tuple(map(key_n3, t)) for t in premises)

        added = []
        for result, premises, rule in sorted(candidates, key=sort_key):
            key = canonical(result)
            if key in refs:
                continue
            leaf_union = ordered_set([leaf for t in premises for leaf in leaves[canonical(t)]])
            depth = 1 + max(depths[canonical(t)] for t in premises)
            steps.append(
                {
                    "ordinal": len(steps),
                    "rule": rule,
                    "result": result,
                    "premises": [refs[canonical(t)] for t in premises],
                    "depth": depth,
                    "orderedDirectInputTriples": premises,
                    "leafUnion": leaf_union,
                }
            )
            refs[key] = {"kind": "step", "ordinal": len(steps) - 1}
            leaves[key], depths[key] = leaf_union, depth
            added.append(result)
        if not added:
            return steps
        closure = ordered_set(closure + added)


def legacy_expected(dataset):

def with_leaf_basis(computed, graph, steps):
    proof = {canonical(t): [t] for t in graph}
    proof.update({canonical(step["result"]): step["leafUnion"] for step in steps})
    for match in computed["matches"]:
        match["leafBasis"] = ordered_set(
            [leaf for triple in match["when"] for leaf in proof[canonical(triple)]]
        )
        match["whenBasis"] = [proof[canonical(triple)] for triple in match["when"]]
        match["sourcePolicy"] = (
            "all original leaf/source occurrences bound to actual input preimage; raw token"
            " spans unavailable where not supplied, never fabricated"
        )
    return computed


def scope_rule_basis(matches, triple):
    bases = [m["leafBasis"] for m in matches if any(h["triple"] == triple for h in m["heads"])]
    if not bases or len({canonical(basis) for basis in bases}) != 1:
        raise ValueError("scope fixture requires one independently specified leaf basis")
    return bases[0]

def record_math(dataset):
    if dataset.get("kind") == "turtle":
        graph = turtle(dataset["text"])
        rules = []
        raw_records = None
    else:
        graph = ordered_set([[record_term(t) for t in row] for row in dataset["records"]])
        rules = dataset["rules"]
        raw_records = dataset["records"]
    rdfs = finite_rdfs(graph)
    computed = with_leaf_basis(
        declared_rules(rules, ordered_set(graph + rdfs)), graph, finite_witnesses(graph)
    )
    witnesses = []
    closure = ordered_set(graph + rdfs)
    for result in rdfs:
        subject, predicate, obj = result
        candidates = []
        for axiom in closure:
            a, p, b = axiom
            if predicate == iri(RDF + "type"):
                if p == iri(RDFS + "domain") and b == obj:
                    candidates += [
                        {"rule": "rdfs2", "premises": [axiom, fact]}
                        for fact in closure
                        if fact[0] == subject and fact[1] == a
                    ]
                if p == iri(RDFS + "range") and b == obj and subject["kind"] == "iri":
                    candidates += [
                        {"rule": "rdfs3", "premises": [axiom, fact]}
                        for fact in closure
                        if fact[2] == subject and fact[1] == a
                    ]
                if p == iri(RDFS + "subClassOf") and b == obj:
                    candidates += (
                        [{"rule": "rdfs9", "premises": [[subject, iri(RDF + "type"), a], axiom]}]
                        if [subject, iri(RDF + "type"), a] in closure
                        else []
                    )
            if predicate == iri(RDFS + "subClassOf") and p == predicate and a == subject:
                right = [b, predicate, obj]
                if right in closure:
                    candidates.append({"rule": "rdfs11", "premises": [axiom, right]})
        witnesses.append(
            {
                "result": result,
                "finiteValidOrderedPremiseChoices": ordered_set(candidates),
                "depthRule": "1+max(depth(premises)); original leaf depth0; no forward/self reference",
                "leafRule": (
                    "exact deduplicated union of all original leaf premises; ever"
                    "y input source occurrence retained separately"
                ),
            }
        )
    return {
        "inputGraph": graph,
        "rdfsAdded": rdfs,
        **computed,
        "witnessConstraints": witnesses,
        "selectedDirectSteps": finite_witnesses(graph),
        "rawTermSourceOccurrences": raw_records,
        "rawAliases": (
            "RDF1.1 string and case-insensitive language identities coalesce only as RD"
            "F identity; raw lexical/datatype/language/source occurrences including Non"
            "e/empty preserved exactly"
        ),
        "rootOccurrences": [
            {
                "triple": head["triple"],
                "ref": {"kind": "ruleHead", "firingIndex": i, "headIndex": head["headIndex"]},
            }
            for i, firing in enumerate(computed["matches"])
            for head in firing["heads"]
        ],
    }

def evaluate_relation(rule, model):
    """Finite predicate self-tests on owned semantic models, never observed runs.

    A genuine caller-capture receipt is a separate required trust gate. Supplying
    equal JSON values cannot satisfy same-issued-object or prove its provenance.
    """

    def at(path):
        current = model
        for key in path:
            current = current[key]
        return current

    op = rule["op"]
    if op == "sha256-equals":
        raw, sha = at(rule["bytes"]), at(rule["digest"])
        if type(raw) is not bytes or type(sha) is not str or digest(raw) != sha:
            raise ValueError("received byte digest differs")
    elif op == "concat-utf8-equals":
        rows, raw = at(rule["rows"]), at(rule["bytes"])
        if type(rows) not in (list, tuple) or len(rows) > rule["maxItems"]:
            raise ValueError("rows type or full count")
        if not all(type(row) is str for row in rows) or "".join(rows).encode() != raw:
            raise ValueError("row tail/encoding differs")
    elif op == "byte-sum-at-most":
        parts = [at(path) for path in rule["parts"]]
        if not all(type(part) is bytes for part in parts) or sum(map(len, parts)) > rule["cap"]:
            raise ValueError("complete wire byte cap")
    elif op == "exact-model":
        assert_exact_model(at(rule["left"]), at(rule["right"]))
    elif op == "declared-allocation-exact":
        expected = declared_allocation_expectation(at(rule["ruleIds"]), at(rule["disabledRuleIds"]))
        assert_exact_model(expected, at(rule["declaration"]))
    elif op == "same-issued-object":
        left, right = at(rule["left"]), at(rule["right"])
        if type(left) in (
            type(None),
            bool,
            int,
            float,
            str,
            bytes,
            dict,
            list,
            tuple,
            set,
            frozenset,
        ):
            raise ValueError("missing/JSON value is not an issued typed object")
        if left is not right:
            raise ValueError("not the same issued caller object")
    else:
        raise ValueError("unknown predicate operator")
    return True


def normalized_source_local(local):
    if type(local) is not str:
        raise ValueError("source local path must be a string")
    prefix = str(EVIDENCE.relative_to(ROOT)) + "/"
    relative = local[len(prefix) :] if local.startswith(prefix) else local
    path = Path(relative)
    allowed = {"inputs/" + n for n in PINS} | {"guidance/" + n for n in DOC_PINS}
    if path.is_absolute() or str(path) != relative or relative not in allowed:
        raise ValueError("source outside declared local identities: " + local)
    target = EVIDENCE / path
    if target.is_symlink() or not target.resolve().is_relative_to(EVIDENCE.resolve()):
        raise ValueError("nonlocal source identity")
    return prefix + relative

            type(row) is not dict
            or not {"local", "bytes", "sha256"}.issubset(row)
            or type(row["bytes"]) is not int
            or row["bytes"] < 0
            or type(row["sha256"]) is not str
            or re.fullmatch(r"[0-9a-f]{64}", row["sha256"]) is None
        ):
            raise ValueError("incomplete source identity record")
        item = copy.deepcopy(row)
        item["local"] = normalized_source_local(item["local"])
        identity = (item["bytes"], item["sha256"])
        if item["local"] in identities and identities[item["local"]] != identity:
            raise ValueError("conflicting source identities: " + item["local"])
        identities[item["local"]] = identity
        raw = (ROOT / item["local"]).read_bytes()
        if (len(raw), digest(raw)) != identity:
            raise ValueError("source registry identity mismatch: " + item["local"])
        if "ref" in item and (
            type(item["ref"]) is not str or re.fullmatch(r"[0-9a-f]{40}", item["ref"]) is None
        ):
            raise ValueError("invalid declared historical ref")
        item["originRefStatus"] = (
            "provided metadata, not independently authenticated"
            if "ref" in item
            else "not provided; pinned local identity only"
        )
        name = Path(item["local"]).name
        item["consumption"] = (
            "retained-reference-only; no exact current observation/input mapping"
            if name == "explanation-goldens.json"
            else "conditional predicates and metadata roles in expectedAuthorityConsumption"
            if name == "consumer-C-goldens.json"
            else "immutable original operation context, not an engine invocation"
            if name == "product-cases.json"
            else "sealed input origin, not expected"
            if name in (ASSET, DESCRIPTOR)
            else (
                "fixed finite expected or normative/interface authority; see per-observatio"
                "n expectedSourceRefs"
            )
        )
        result.append(item)
    return result

def compare_golden_report(report, expected):
    consumed = []
    if "conforms" in expected:
        assert_exact_model(expected["conforms"], report["conforms"])
        consumed.append("conforms")
    if "components" in expected:
        actual = sorted({v["constraint"]["value"].removeprefix(SH) for v in report["violations"]})
        assert_exact_model(sorted(expected["components"]), actual)
        consumed.append("components")
    if "requiredResultRows" in expected:
        mapping = {
            "focusNode": "focus",
            "resultPath": "path",
            "value": "value",
            "sourceShape": "sourceShape",
            "sourceConstraintComponent": "constraint",
            "resultSeverity": "severity",
        }
        rows = [
            {
                key: None if value.get(name) is None else n3_term(value[name])
                for name, key in mapping.items()
            }
            for value in expected["requiredResultRows"]
        ]
        assert_exact_model(sorted(rows, key=canonical), sorted(report["violations"], key=canonical))
        consumed.append("requiredResultRows")
    return consumed

def validate_raw_term_domains(entry, dataset):
    expected = entry["expected"]
    math = expected.get("businessMath")
    if math is not None:
        raw = math["rawTermSourceOccurrences"] if dataset.get("kind") != "profile" else None
        if dataset.get("kind") != "profile":
            supplied = dataset.get("records")
            assert_exact_model(supplied, raw)
            if raw is not None:
                for row in raw:
                    if type(row) is not list or len(row) != 3:
                        raise ValueError("raw record must retain three term occurrences")
                    for term in row:
                        if type(term) is not dict or set(term) != set(
                            FIELD_POLICIES["RawTermSourceOccurrence"]["exactKeys"]
                        ):
                            raise ValueError("raw term exact keys differ")
                        if (
                            term["kind"] not in ("iri", "literal")
                            or type(term["lexical_form"]) is not str
                        ):
                            raise ValueError("raw term scalar type differs")
                        if any(
                            term[key] is not None and type(term[key]) is not str
                            for key in ("datatype_iri", "language_raw")
                        ):
                            raise ValueError("raw term nullable string differs")
    elif "records" in dataset and entry["operation"] in ("fullview", "dag", "restore-negative"):
        raise ValueError("structured observation lost its complete businessMath")

    def walk(value):
        if isinstance(value, dict):
            term_shapes = [
                {"kind", "value"},
                {"kind", "lexical", "datatype", "language"},
                {"kind", "label"},
            ]
            kind = value.get("kind")
            if (type(kind) is str and kind in ("iri", "literal", "blank")) or set(
                value
            ) in term_shapes:
                if type(kind) is not str or kind not in ("iri", "literal", "blank"):
                    raise ValueError("unknown canonical RDF term kind")
                keys = {
                    "iri": {"kind", "value"},
                    "literal": {"kind", "lexical", "datatype", "language"},
                    "blank": {"kind", "label"},
                }[kind]
                if set(value) != keys:
                    raise ValueError("canonical RDF term domain differs")
            for key, child in value.items():
                if key != "rawTermSourceOccurrences":
                    walk(child)
        elif isinstance(value, list):
            for child in value:
                walk(child)

    walk(expected)

def validate_oracle(oracle, asset, descriptor):
    entries = oracle["observations"]
    if [e["id"] for e in entries] != [e["id"] for e in asset["observations"]]:
        raise ValueError("missing/reordered/extra observation")
    if oracle["caseSets"] != descriptor["caseSets"]:
        raise ValueError("case sets changed")
    for expected, actual in zip(asset["observations"], entries):
        if (
            actual["projectionSha256"] != expected["projectionSha256"]
            or actual["parameters"] != expected["parameters"]
        ):
            raise ValueError("input binding changed")
        if actual["oracleStatus"] != ("pending" if actual["pendingFields"] else "specified"):
            raise ValueError("pending hidden")
        validate_raw_term_domains(actual, asset["datasets"][expected["datasetId"]])
        if actual["operation"] == "scope" and actual["expected"]["semanticState"] == "normal":
            for source in actual["expected"]["factSources"]:
                if source["role"] == "rule":
                    assert_exact_model(
                        scope_rule_basis(actual["expected"]["matches"], source["triple"]),
                        source["basis"],
                    )
        declaration = allocation_for_observation(expected, asset["datasets"][expected["datasetId"]])
        assert_exact_model(declaration, actual["declaredAllocationExpectation"])
        if (
            declaration is not None
            and declaration["status"] == "pending"
            and not any(
                item["field"] == "declaredAllocationExpectation" for item in actual["pendingFields"]
            )
        ):
            raise ValueError("declaration metadata gap hidden")
        result = actual["expected"]
        if result["semanticState"] not in ("normal", "known-rejection", "not-executed"):
            raise ValueError("unknown promoted to matched")
        if result["semanticState"] == "known-rejection" and result.get("report") is not None:
            raise ValueError("rejection has partial report")
        if "report" in result and result["report"] is not None:
            report = result["report"]
            if set(report) != {"conforms", "violations"} or type(report["conforms"]) is not bool:
                raise ValueError("report schema")
            for row in report["violations"]:
                if set(row) != set(FIELD_POLICIES["SHACL"]["violationExactKeys"]):
                    raise ValueError("incomplete SHACL six fields")
            if report["conforms"] != (not report["violations"]):
                raise ValueError("report conforms inconsistent")
    if oracle["wholeOracleComplete"] != all(e["oracleStatus"] == "specified" for e in entries):
        raise ValueError("false whole oracle completion")
    assert_exact_model(
        expected_authority_consumption(entries), oracle["expectedAuthorityConsumption"]
    )
    return True


def assert_exact_model(expected, actual):
    """Own-model mutation tests only; never consumes product JSONL or launches it."""
    if type(expected) is not type(actual):
        raise ValueError("exact model type differs")
    if isinstance(expected, dict):
        if expected.keys() != actual.keys():
            raise ValueError("exact model fields differ")
        for key in expected:
            assert_exact_model(expected[key], actual[key])
    elif isinstance(expected, list):
        if len(expected) != len(actual):
            raise ValueError("model tail/multiplicity differs")
        for a, b in zip(expected, actual):
            assert_exact_model(a, b)
    elif expected != actual:
        raise ValueError("model scalar differs")
    return True


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--check",
        action="store_true",
        help="rebuild in memory and require exact committed data bytes",
    )
    args = parser.parse_args(argv)
    artifacts = construct()
    validate_oracle(artifacts["oracle.json"], read_pinned(ASSET), read_pinned(DESCRIPTOR))
    if not args.check:
        OUTPUT.mkdir(parents=True, exist_ok=True)
    rows = []
    for name, value in artifacts.items():
        raw = (
            json.dumps(value, ensure_ascii=True, sort_keys=True, indent=2, allow_nan=False).encode()
            + b"\n"
        )
        target = OUTPUT / name
        if args.check:
            if not target.exists() or target.read_bytes() != raw:
                raise ValueError("non-reproducible oracle artifact: " + name)
        else:
            target.write_bytes(raw)
        rows.append(
            {"path": str(target.relative_to(ROOT)), "bytes": len(raw), "sha256": digest(raw)}
        )
    coverage = artifacts["coverage.json"]
    print(
        json.dumps(
            {
                "mode": "check" if args.check else "build",
                "observations": 417,
                "datasets": 320,
                "statuses": coverage["byStatus"],
                "wholeOracleComplete": coverage["wholeOracleComplete"],
                "artifacts": rows,
            },
            sort_keys=True,
        )
    )
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
