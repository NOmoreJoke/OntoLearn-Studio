import { readFileSync, writeFileSync } from 'node:fs'

const ts = JSON.parse(readFileSync('/tmp/i03-r2-minimax/probe/results-ts.json', 'utf8'))
const py = JSON.parse(readFileSync('/tmp/i03-r2-minimax/probe/results-py.json', 'utf8'))
const tsMap = new Map(ts.map((r) => [r.id, r]))
const pyMap = new Map(py.map((r) => [r.id, r]))

const divergences = []
const ids = new Set([...tsMap.keys(), ...pyMap.keys()])
for (const id of ids) {
  const a = tsMap.get(id)
  const b = pyMap.get(id)
  if (!a || !b) {
    divergences.push({ id, ts: a ?? 'MISSING', py: b ?? 'MISSING' })
    continue
  }
  if (a.result !== b.result) divergences.push({ id, ts: a.result, py: b.result, tsError: a.error, pyError: b.error })
}
const tsThrows = ts.filter((r) => r.result === 'THROW')
const pyThrows = py.filter((r) => r.result === 'THROW')
writeFileSync(
  '/tmp/i03-r2-minimax/probe/divergences.json',
  JSON.stringify({ total: ids.size, divergences, tsThrows, pyThrows }, null, 1),
)
console.log('compared', ids.size, 'divergences', divergences.length)
console.log('ts throws', tsThrows.length, 'py throws', pyThrows.length)
for (const d of divergences.slice(0, 40)) console.log('DIVERGE', JSON.stringify(d))
